3. PK JAWAB M2W
	select count(distinct(btmnobuk)) 
	from cad 
	left join btm on btmdata ='BUKUTAMU' and btmlokjual = cadkdcab and btmsts in ('CLSD','724','721','720',
		'716','631','727','3502','714','712','INPG','713','719','711','TACC','715') 
		and btmdvsjl = 'M2W' and btmtgl = {ltgl2} 
	left join akt as akt631 on akt631.aktkode = btmnobuk and akt631.aktdata = 'HISTORISDM' 
		and akt631.akttype = '631' and akt631.aktprm in (select min(akt631a.aktprm) 
			from akt as akt631a 
			where akt631a.aktkode = btmnobuk and akt631a.aktdata = 'HISTORISDM' 
				and akt631a.akttype = '631') 
	left join akt as akt714 on akt714.aktkode = btmnobuk and akt714.aktdata = 'HISTORISDM' 
		and akt714.akttype = '714' 
		and akt714.aktprm in (select max(akt714a.aktprm) 
			from akt as akt714a 
			where akt714a.aktkode = btmnobuk and akt714a.aktdata = 'HISTORISDM' 
				and akt714a.akttype = '714') 
        where caddata = 'TGTMKTM2W' and cadkode = '2403' and cadtype = 'TGTMKTM2W' and cadkdcab = {cadkdcab} 
		and btmnobuk <> '' and akt714.aktkode is null;
